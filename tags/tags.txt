# IELTS Content Tag Vocabulary — 3-Layer System
# ══════════════════════════════════════════════════════════════════
# Layer 1  — 4 fixed category tags (always exactly one per topic)
# Layer 2  — thematic cluster tags (domain / sub-domain)
# Layer 3  — specific / concrete tags (narrow concept under an L2)
# Qualifiers — describe tone/quality of a topic; stored separately
#              in qualifier_tags field, not in content_tags
# ══════════════════════════════════════════════════════════════════

## ── LAYER 1 — fixed categories ──────────────────────────────────
people
place
object
experience/activity

## ── LAYER 2 & 3 — by L1 parent ──────────────────────────────────

# [people]
# L2
family              # family member, relatives, family gathering, family business context
friendship          # friend, close companion, trust, personal bond
celebrity           # famous person, public figure, well-known personality
influence           # persuasion, encouragement, motivated someone to change behaviour
admiration          # admire, look up to, inspirational, in awe of someone
talent              # natural gift, innate skill, born ability
intelligence        # smart, clever, sharp mind, shrewd thinking
happiness           # smiling, joyful, cheerful, positive emotional state
child               # a child or children as the subject (age-specific)
# L3 under people L2s
conflict_resolution # [family] apology, saying sorry, making up after a conflict, forgiveness
helping_others      # [friendship] kindness, generosity, volunteering, being helpful to people
collaboration       # [friendship] working together, joint effort, group cooperation

# [place]
# L2
home                # home life, at home, domestic situation, where you live
travel              # journey, trip, vacation, visiting places, going somewhere
nature              # outdoor, natural settings, scenic views, parks, mountains, wilderness
architecture        # building, structure, design, construction, landmark
# L3 under place L2s
everyday_life       # [home] essential, can't live without, daily necessity, routine dependency
adventure           # [travel] exciting, thrilling, bold activity, daring new experience
international       # [travel] foreign country, abroad, overseas, cross-cultural experience
navigation          # [travel] lost, finding your way, directions, getting around
animals             # [nature] wild animal, creature, wildlife, fauna, pet
conservation        # [nature] caring for/protecting the environment or animals; eco-minded behaviour

# [object]
# L2
books               # book as object, written publication
heirloom            # important old thing kept in family, treasured possession, keepsake
toy                 # toy as physical object; childhood plaything
phone               # mobile phone, smartphone; often in context of rules or dependency
money               # spending, overspending, financial decisions, saving, cost
technology          # tech device, gadget, electronic equipment, digital tool; broader than phone
# L3 under object L2s
app                 # [technology] program, software application, digital app, computer/phone app
social_media        # [technology] social media platforms, internet posts, online content

# [experience/activity]
# L2
art                 # drawing, painting, visual arts, artistic creation, gallery
music               # music, singing, concerts, songs, instruments
reading             # act of reading, books as activity
movies              # film, movie, cinema, a film you watched
food                # dinner, meal, eating, cuisine, restaurant
work                # job, career, occupation, part-time work, family business, employment
sports              # sport, physical game, athletic activity, exercise for competition
shopping            # buying items, retail, purchase, customer experience in a shop
learning            # studying, gaining knowledge, education, acquiring a skill
science             # scientific subject, field of science, biology, robotics, research
creativity          # imagination, creative thinking, innovative ideas, inventive approach
culture             # tradition, cultural heritage, customs, folklore background
communication       # talking, expressing oneself, verbal/written interaction
media               # TV, online programs, streaming, entertainment content (non-social)
celebration         # joyful gathering, festival, occasion with many smiling people
disruption          # unexpected interruption, power cut, inconvenience, things going wrong
mistake             # accident, error, broke something, careless action, regret
service             # good/bad service, hospitality, customer care
achievement         # proud of, accomplishment, success, milestone reached
passion             # loves doing something, deeply interested in, devoted to
aspiration          # want to, would like to, dream of, hope to do/have/be something
anticipation        # waiting, looking forward to, expecting something special
childhood           # anything specifically framed in the context of being a child/young
habit               # regular pattern of behaviour, routine, repeated action or practice
# L3 under experience/activity L2s
self-learning       # [learning] self-taught, learned without a teacher, independent study
curiosity           # [learning] want to know more, eager to learn about, interested in exploring
stories             # [culture] traditional story, tale, narrative, folklore
language            # [culture/communication] foreign language, learning/speaking another language
advice              # [communication] giving or receiving guidance, counselling, recommendations
social_event        # [celebration] event, occasion, gathering, party, group situation
first_time          # [celebration] for the first time, debut, new experience, never done before
restriction         # [disruption] not allowed, banned, prohibited, rules imposed on behaviour
planning            # [achievement] making plans, goal-setting, organised, forward-thinking
self-improvement    # [achievement] want to improve, develop a skill, get better at something
decision            # [achievement] making a choice, weighing options, an important decision
problem-solving     # [intelligence→people] solved a problem, found a solution, clever solution
likes_dislikes      # [passion] enjoyed, liked, loved, disliked, didn't enjoy — attitude qualifier
nostalgia           # [childhood] childhood memory, fond memory of the past (use with childhood or heirloom)

## ── QUALIFIERS — stored in qualifier_tags, not content_tags ─────
# Qualifiers describe the tone, emotional quality, or angle of a
# topic. They do not belong to the L1→L2→L3 hierarchy.
memorable           # special, noteworthy, stands out in memory, significant moment
sentimental         # emotionally attached, personally meaningful, nostalgic feeling
peaceful            # quiet, calm, tranquil — quality of a place or moment
useful              # helpful, practical, beneficial — quality of an object or experience
interesting         # unusual, thought-provoking, captured attention — general interest qualifier
